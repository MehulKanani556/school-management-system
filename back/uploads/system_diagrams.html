Entity Relationship Model

erDiagram
    User ||--o| Student : "has profile"
    User ||--o| Teacher : "has profile"
    User {
        string id PK
        string firstName
        string lastName
        string email
        string role
        string password
    }
    
    School ||--o{ Student : "enrolls"
    School ||--o{ Teacher : "employs"
    School {
        string id PK
        string name
        string address
    }

    Student }|--o| ClassSection : "assigned to"
    Student }|--o| Standard : "belongs to"
    Student {
        string id PK
        string admissionNumber
        string rollNumber
        string firstName
        string lastName
        string email
    }

    Teacher {
        string id PK
        string userId FK
        string employeeId
        string firstName
        string lastName
        string email
    }

    ClassSection }|--|| Standard : "belongs to"
    ClassSection }|--|| Teacher : "has class teacher"
    ClassSection {
        string id PK
        string standardId FK
        string classTeacher FK
        string sectionLabel
    }

    Standard {
        string id PK
        int level
        string name
    }

    Attendance }|--|| ClassSection : "recorded for"
    Attendance }|--|| User : "submitted by"
    Attendance ||--|{ AttendanceRecord : "contains"
    Attendance {
        string id PK
        string schoolId FK
        string standardId FK
        string classSection FK
        string date
        string submittedBy FK
    }

    AttendanceRecord }|--|| Student : "references"
    AttendanceRecord {
        string studentId FK
        string status
    }

    BehaviorLog }|--|| Student : "concerns"
    BehaviorLog }|--|| Teacher : "logged by"
    BehaviorLog {
        string id PK
        string studentId FK
        string teacherId FK
        string type
        string category
        string description
        string actionTaken
    }
